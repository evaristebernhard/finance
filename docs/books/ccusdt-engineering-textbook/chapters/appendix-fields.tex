\chapter{字段词典}

\section{decision frame 字段}

\begin{longtable}{p{0.28\textwidth}p{0.20\textwidth}p{0.42\textwidth}}
\toprule
字段 & 层 & 解释 \\
\midrule
\code{schema_id} & cache & schema 标识 \\
\code{runtime_safe} & cache & 是否声明为 runtime-safe \\
\code{observed_seq} & clock & 当前观察事件序号 \\
\code{exchange_ts_us} & clock & 交易所时间戳 \\
\code{local_ts_us} & clock & 本地时间戳 \\
\code{event_index} & clock & decision frame 事件索引 \\
\code{best_bid_price} & quote/L2 & 当前最好买价 \\
\code{best_bid_amount} & quote/L2 & 当前最好买价数量 \\
\code{best_ask_price} & quote/L2 & 当前最好卖价 \\
\code{best_ask_amount} & quote/L2 & 当前最好卖价数量 \\
\code{mid_price} & quote/L2 & \((bid+ask)/2\) \\
\code{spread_bps} & quote/L2 & \((ask-bid)/mid \times 10000\) \\
\code{trade_window_count} & trade & 当前 rolling window 内 trade 数 \\
\code{trade_buy_amount} & trade & buy 主动成交量 \\
\code{trade_sell_amount} & trade & sell 主动成交量 \\
\code{trade_flow_imbalance} & trade & \((buy-sell)/(buy+sell)\) \\
\code{frames_since_mid_change} & path & mid 连续未变化 frame 数 \\
\code{past_event_25_bps} & path & 过去约 25 event 的 mid log return bps \\
\code{fwd_time_60s_bucket} & clock & 60s bucket，用于限制重复触发 \\
\bottomrule
\end{longtable}

\begin{longtable}{p{0.30\textwidth}p{0.60\textwidth}}

\section{exit 输出字段}

\begin{longtable}{p{0.30\textwidth}p{0.60\textwidth}}
\toprule
字段 & 解释 \\
\midrule
\code{exit_ts_us} & 退出时间 \\
\code{exit_profile} & \code{fixed60_mid}、\code{fixed60_taker}、manager path 等 \\

\code{exit_ask} & 退出 ask \\

\code{net_bps} & 扣 fee/pressure/slippage 或 taker 后口径 \\
\code{weighted_bps} & 用 actual exposure 加权后的贡献 \\
\bottomrule
\end{longtable}
